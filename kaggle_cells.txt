# Server ကို Background တွင် run ပြီး Cloudflare Tunnel မှ Public Link ထုတ်ပေးခြင်း
import subprocess
import time
import re
import os
import socket

# 0. Kaggle Internet Connection စစ်ဆေးခြင်း
def check_internet():
    try:
        socket.create_connection(("1.1.1.1", 53), timeout=3)
        return True
    except Exception:
        return False

if not check_internet():
    print("❌ သတိပေးချက်: Kaggle တွင် Internet ပိတ်ထားပါသည် (Internet is OFF)!")
    print("👉 ညာဘက်ခြမ်းရှိ 'Notebook options' (Settings ⚙️) သို့ သွားပါ")
    print("👉 'Internet' ကို 'Internet ON' အဖြစ် ဖွင့်ပေးပါ (Phone verification လိုအပ်နိုင်ပါသည်)")
    print("👉 ပြီးနောက် ဤ Cell ကို ပြန်လည် Run ပေးပါ\n")

# 1. Start FastAPI Server in background
print("🚀 Starting FastAPI Studio Server on port 8000...")
!nohup uvicorn app.main:app --host 0.0.0.0 --port 8000 > server.log 2>&1 &

time.sleep(3)

# 2. Start Cloudflare Tunnel (Logs redirected to tunnel.log to prevent pipe blocking)
print("🌐 Creating Public Cloudflare Tunnel...")
!rm -f tunnel.log
!nohup ./cloudflared tunnel --url http://127.0.0.1:8000 > tunnel.log 2>&1 &

# 3. Extract and display public link
public_url = None
start_time = time.time()

while time.time() - start_time < 30:
    time.sleep(1)
    if os.path.exists("tunnel.log"):
        with open("tunnel.log", "r", encoding="utf-8", errors="ignore") as f:
            content = f.read()
            match = re.search(r"https://[a-zA-Z0-9-]+\.trycloudflare\.com", content)
            if match:
                public_url = match.group(0)
                break

print("\n" + "=" * 65)
if public_url:
    print("🎉 RECAP FREE STUDIO IS NOW LIVE & ACCESSIBLE PUBLICLY!")
    print(f"👉 PUBLIC WEB LINK: {public_url}")
    print("=" * 65)
    print("\n💡 ဖုန်း (Mobile) သို့မဟုတ် ကွန်ပျူတာ browser မှ အထက်ပါ link ကို နှိပ်၍ အသုံးပြုနိုင်ပါပြီ။")
    print("⚡ NVIDIA T4 GPU Hardware Acceleration အပြည့်အဝ အသုံးပြုနိုင်ပါမည်။")
    print("⚠️ Notebook ကို Run ထားသရွေ့ Server အလုပ်လုပ်နေပါမည်။")
else:
    print("⚠️ Tunnel URL ထွက်ပေါ်ရန် စောင့်ဆိုင်းနေပါသည် သို့မဟုတ် အမှားတစ်ခု ဖြစ်ပေါ်နေပါသည်...")
    print("--- [TUNNEL LOG OUTPUT] ---")
    if os.path.exists("tunnel.log"):
        with open("tunnel.log", "r", encoding="utf-8", errors="ignore") as f:
            print(f.read()[-1000:])
    print("--- [SERVER LOG OUTPUT] ---")
    if os.path.exists("server.log"):
        with open("server.log", "r", encoding="utf-8", errors="ignore") as f:
            print(f.read()[-500:])
print("=" * 65)

# Keep cell alive
try:
    while True:
        time.sleep(10)
except KeyboardInterrupt:
    print("\nStopping Server and Tunnel...")

    !pkill -f cloudflared
